//+------------------------------------------------------------------+ //| TradingTerminalPanel.mq5 | //| Copyright 2026, Tu Nombre | //+------------------------------------------------------------------+ #property copyright "Trading Terminal AI" #property indicator_chart_window #property indicator_plots 0 //--- Parámetros de entrada input group "=== Configuración Panel ===" input int InpPanelWidth = 260; // Ancho del panel (px) input color InpBgColor = C'19,23,34'; // Color Fondo Panel input color InpBorderColor = C'41,98,255';// Color Borde Principal input color InpTextColor = C'209,212,220'; // Color Texto input group "=== Configuración VSA ===" input int InpVSAPeriod = 20; // Período Promedio Volumen input double InpVSAFactor = 1.8; // Multiplicador Ballena //--- Variables Globales int panelX = 10; int panelY = 10; string prefix = "TTP_"; //+------------------------------------------------------------------+ //| Custom indicator initialization function | //+------------------------------------------------------------------+ int OnInit() { // 1. Desplazar gráfico a la derecha para dejar margen a la escala ChartSetInteger(0, CHART_SHIFT, true); ChartSetDouble(0, CHART_SHIFT_SIZE, 15.0); // 15% de espacio // 2. Crear elementos gráficos del panel CreateDashboardUI(); // 3. Primera actualización de datos UpdateDashboardData(); return(INIT_SUCCEEDED); } //+------------------------------------------------------------------+ //| Custom indicator deinitialization function | //+------------------------------------------------------------------+ void OnDeinit(const int reason) { // Limpiar todos los objetos creados al quitar el indicador ObjectsDeleteAll(0, prefix); ChartRedraw(0); } //+------------------------------------------------------------------+ //| Custom indicator iteration function | //+------------------------------------------------------------------+ int OnCalculate(const int rates_total, const int prev_calculated, const datetime &time[], const double &open[], const double &high[], const double &low[], const double &close[], const long &tick_volume[], const long &volume[], const int &spread[]) { // Actualizar datos del panel en cada nuevo tick UpdateDashboardData(); return(rates_total); } //+------------------------------------------------------------------+ //| Dibuja la estructura UI del Panel Lateral | //+------------------------------------------------------------------+ void CreateDashboardUI() { int currentY = panelY; // --- Fondo del Panel CreateRectangle(prefix + "BG", panelX, currentY, InpPanelWidth, 420, InpBgColor, InpBorderColor); // --- Encabezado PROYECCIÓN ALGORÍTMICA AI CreateLabel(prefix + "AI_Title", "PROYECCIÓN ALGORÍTMICA AI", panelX + 10, currentY + 10, C'0,229,255', 9, true); CreateLabel(prefix + "AI_Live", "LIVE", panelX + InpPanelWidth - 40, currentY + 10, C'0,230,118', 8, true); currentY += 30; // --- Tarjetas de Sesgo por Timeframe (15M, 1H, 4H) int boxW = (InpPanelWidth - 30) / 3; CreateBoxTF("15M", panelX + 10, currentY, boxW, "p_15m"); CreateBoxTF("1H", panelX + 15 + boxW, currentY, boxW, "p_1h"); CreateBoxTF("4H", panelX + 20 + (boxW*2), currentY, boxW, "p_4h"); currentY += 60; // --- Sección SUPERTREND CreateRectangle(prefix + "ST_Box", panelX + 10, currentY, InpPanelWidth - 20, 45, InpBgColor, C'0,230,118'); CreateLabel(prefix + "ST_Title", "SUPERTREND", panelX + 18, currentY + 6, C'0,230,118', 8, true); CreateLabel(prefix + "ST_Sig_Lbl", "Señal:", panelX + 18, currentY + 24, InpTextColor, 8); CreateLabel(prefix + "ST_Signal", "CALC...", panelX + 70, currentY + 24, C'255,215,0', 8, true); CreateLabel(prefix + "ST_Prc_Lbl", "Precio:", panelX + 130, currentY + 24, InpTextColor, 8); CreateLabel(prefix + "ST_Price", "--", panelX + 175, currentY + 24, InpTextColor, 8, true); currentY += 55; // --- Tabla VSA Multi-Timeframe CreateLabel(prefix + "VSA_Title", "ANÁLISIS VSA MULTI-TIMEFRAME", panelX + 10, currentY, C'41,98,255', 8, true); currentY += 18; // Encabezados Tabla CreateLabel(prefix + "TH_TF", "TF", panelX + 15, currentY, C'120,123,134', 8, true); CreateLabel(prefix + "TH_VSA", "VSA", panelX + 50, currentY, C'120,123,134', 8, true); CreateLabel(prefix + "TH_VOL", "Volumen", panelX + 100, currentY, C'120,123,134', 8, true); CreateLabel(prefix + "TH_BUY", "% Compra", panelX + 180, currentY, C'0,230,118', 8, true); currentY += 15; // Filas MTF ENUM_TIMEFRAMES tfs[4] = {PERIOD_M5, PERIOD_M15, PERIOD_H1, PERIOD_H4}; string tfNames[4] = {"5m", "15m", "1h", "4h"}; for(int i = 0; i < 4; i++) { int rowY = currentY + (i * 18); CreateLabel(prefix + "TD_TF_" + tfNames[i], tfNames[i], panelX + 15, rowY, C'255,255,255', 8, true); CreateLabel(prefix + "TD_VSA_" + tfNames[i], "NEU", panelX + 50, rowY, C'120,123,134', 8, true); CreateLabel(prefix + "TD_VOL_" + tfNames[i], "--", panelX + 100, rowY, InpTextColor, 8); CreateLabel(prefix + "TD_BUY_" + tfNames[i], "--%", panelX + 180, rowY, C'0,230,118', 8); } ChartRedraw(0); } //+------------------------------------------------------------------+ //| Actualiza los cálculos en tiempo real | //+------------------------------------------------------------------+ void UpdateDashboardData() { ENUM_TIMEFRAMES tfs[4] = {PERIOD_M5, PERIOD_M15, PERIOD_H1, PERIOD_H4}; string tfNames[4] = {"5m", "15m", "1h", "4h"}; for(int i = 0; i < 4; i++) { MqlRates rates[]; ArraySetAsSeries(rates, true); if(CopyRates(_Symbol, tfs[i], 0, InpVSAPeriod + 1, rates) < InpVSAPeriod) continue; // Cálculo simplificado de volumen VSA double avgVol = 0; for(int j = 1; j <= InpVSAPeriod; j++) avgVol += (double)rates[j].tick_volume; avgVol /= InpVSAPeriod; double currentVol = (double)rates[0].tick_volume; double range = rates[0].high - rates[0].low; double buyPct = (range > 0) ? ((rates[0].close - rates[0].low) / range) * 100.0 : 50.0; // Determinar Estado VSA string vsaStatus = "NEU"; color vsaColor = C'120,123,134'; if(currentVol > (avgVol * InpVSAFactor)) { if(buyPct > 55.0) { vsaStatus = "BUY"; vsaColor = C'0,230,118'; } else if(buyPct < 45.0) { vsaStatus = "SELL"; vsaColor = C'255,23,68'; } } // Actualizar Labels UI ObjectSetString(0, prefix + "TD_VSA_" + tfNames[i], OBJPROP_TEXT, vsaStatus); ObjectSetInteger(0, prefix + "TD_VSA_" + tfNames[i], OBJPROP_COLOR, vsaColor); ObjectSetString(0, prefix + "TD_VOL_" + tfNames[i], OBJPROP_TEXT, DoubleToString(currentVol, 0)); ObjectSetString(0, prefix + "TD_BUY_" + tfNames[i], OBJPROP_TEXT, DoubleToString(buyPct, 1) + "%"); } // Actualizar lectura rápida Supertrend (Basado en precio/cierre actual) double closeCurr = iClose(_Symbol, _Period, 0); double closePrev = iClose(_Symbol, _Period, 1); if(closeCurr > closePrev) { ObjectSetString(0, prefix + "ST_Signal", OBJPROP_TEXT, "BUY"); ObjectSetInteger(0, prefix + "ST_Signal", OBJPROP_COLOR, C'0,230,118'); } else { ObjectSetString(0, prefix + "ST_Signal", OBJPROP_TEXT, "SELL"); ObjectSetInteger(0, prefix + "ST_Signal", OBJPROP_COLOR, C'255,23,68'); } ObjectSetString(0, prefix + "ST_Price", OBJPROP_TEXT, DoubleToString(closeCurr, _Digits)); ChartRedraw(0); } //+------------------------------------------------------------------+ //| Funciones Auxiliares para creación de Objetos Gráficos | //+------------------------------------------------------------------+ void CreateRectangle(string name, int x, int y, int w, int h, color bgColor, color borderColor) { ObjectCreate(0, name, OBJ_RECTANGLE_LABEL, 0, 0, 0); ObjectSetInteger(0, name, OBJPROP_YSIZE, h); ObjectSetInteger(0, name, OBJPROP_BGCOLOR, bgColor); ObjectSetInteger(0, name, OBJPROP_COLOR, borderColor); ObjectSetInteger(0, name, OBJPROP_BORDER_TYPE, BORDER_FLAT); ObjectSetInteger(0, name, OBJPROP_SELECTABLE, false); } void CreateLabel(string name, string text, int x, int y, color txtColor, int fontSize = 8, bool bold = false) { ObjectCreate(0, name, OBJ_LABEL, 0, 0, 0); ObjectSetInteger(0, name, OBJPROP_CORNER, CORNER_LEFT_UPPER); ObjectSetInteger(0, name, OBJPROP_XDISTANCE, x); ObjectSetInteger(0, name, OBJPROP_YDISTANCE, y); ObjectSetString(0, name, OBJPROP_TEXT, text); ObjectSetString(0, name, OBJPROP_FONT, bold ? "Arial Bold" : "Arial"); ObjectSetInteger(0, name, OBJPROP_FONTSIZE, fontSize); ObjectSetInteger(0, name, OBJPROP_COLOR, txtColor); ObjectSetInteger(0, name, OBJPROP_SELECTABLE, false); } void CreateBoxTF(string tfTitle, int x, int y, int w, string id) { CreateRectangle(prefix + "Box_" + id, x, y, w, 45, InpBgColor, C'42,46,57'); CreateLabel(prefix + "LblTF_" + id, tfTitle, x + (w/4), y + 5, C'120,123,134', 7); CreateLabel(prefix + "Val_" + id, "BEAR", x + 8, y + 20, C'255,23,68', 8, true); } //+------------------------------------------------------------------+
